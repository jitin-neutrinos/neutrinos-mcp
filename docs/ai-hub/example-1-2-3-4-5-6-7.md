## Example

```code
import { InferenceSDK } from 'idp-inference-sdk';

const sdk = new InferenceSDK({
  endpoint: 'https://api.aihub.com',
  timeout: 60000,
  trace_oltp_endpoint: 'https://api.aihub.com/traces',
  log_oltp_endpoint: 'https://api.aihub.com/logs',
});

const result = await sdk.classification.text.predict({
  token: 'YOUR_API_TOKEN',
  input: { text: 'Please classify this sentence.' }
});
```
