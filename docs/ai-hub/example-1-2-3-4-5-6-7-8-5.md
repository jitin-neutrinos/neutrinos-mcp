```code
const response: IFileListResponse = {
  total: 1,
  data: [
    {
      _id: "6123456789abcdef01234567",
      file_name: "example.txt",
      size: 123456,
      mime_type: "text/plain",
      file_url: "https://example.com/files/6123456789abcdef01234567",
      file_uuid: "6123456789abcdef01234567",
      created_by: "John Doe",
      uploaded_by: "john@example.com",
      created_at: "2025-07-09T10:00:00.000Z",
      updated_at: "2025-07-09T10:30:00.000Z"
    }
  ]
};

console.log(`Files found: ${response.total}`);
console.log(`First file name: ${response.data[0].file_name}`);
```
