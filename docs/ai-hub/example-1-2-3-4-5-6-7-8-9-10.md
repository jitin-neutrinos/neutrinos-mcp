## Example

```code
const sdk = new InferenceSDK(config);
const modelCatalogService = sdk.modelCatalog.root;

// List models
const models = await modelCatalogService.modelList({
  token: '123',
  training_type: 'Classification',
  data_type: 'Text',
  page_number: 0,
  page_size: 10,
  sort: 'desc'
});

// List model versions
const versions = await modelCatalogService.versionList({
  token: '123',
  training_config_id: '123',
  page_number: 0,
  page_size: 10,
  sort: 'desc'
});

// List environments
const environments = await modelCatalogService.environmentList({
  token: '123',
});

// Generate token
const token = await modelCatalogService.generateToken({
  token: '123',
  training_config_id: '123',
  training_id: '123',
  environment_id: '123',
  expire: TokenExpireEnum.HOUR_1,
});
```
